\begin{frame}{LatentMoE Cost Model}{Router and communication costs decrease more slowly than expert computation}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.47\textwidth}
      {\NvidiaSansMedium Conventional experts}\par
      \vspace{.04in}{\color{NvidiaGreen}\rule{\linewidth}{.6pt}}\par\vspace{.10in}
      \[
        \text{expert work} \propto T \times K \times H \times H_{\mathrm{ffn}}
      \]
      \small Large expert GEMMs amortize routing, launch, and metadata overhead.
    \end{column}
    \begin{column}{.47\textwidth}
      {\NvidiaSansMedium Latent experts}\par
      \vspace{.04in}{\color{NvidiaGreen}\rule{\linewidth}{.6pt}}\par\vspace{.10in}
      \[
        H_{\mathrm{ffn}} \downarrow \quad\Rightarrow\quad \text{expert time} \downarrow
      \]
      \small Router remains $T\times E$; ownership and token exchange remain $T\times K$.
    \end{column}
  \end{columns}
  \vspace{.15in}
  \begin{NvidiaEvidence}{Sparse-path budget}
    The relevant optimization unit is the complete sparse path: selection, compact metadata,
    dispatch preprocessing, communication, expert compute, and combine.
  \end{NvidiaEvidence}
\end{frame}

\begin{frame}[fragile]{Kernel Design}{Coordinated data movement and selection determine router throughput}
  \NvidiaSource{\href{https://github.com/NVIDIA/TransformerEngine/pull/3012}{TE PR \#3012};
    fused-router optimization note}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.51\textwidth}
      \begin{NvidiaCode}
prefetch(tile[0], buffer[0])
for tile in token_tiles:
    prefetch(tile[next], buffer[next])
    wait(buffer[current])
    score(buffer[current])
    radix_topk()
    write_selected()
    swap(current, next)
      \end{NvidiaCode}
    \end{column}
    \begin{column}{.44\textwidth}
      {\NvidiaSansMedium Throughput mechanisms}\par
      \vspace{.04in}{\color{NvidiaGreen}\rule{\linewidth}{.6pt}}\par\vspace{.06in}
      \begin{itemize}
        \item Persistent execution maintains SM occupancy across token tiles.
        \item Double-buffered \texttt{cp.async} overlaps score loads with radix selection.
        \item Packed histograms cut radix counters from 32 to 4; templates remove
          hot-path branches.
        \item Two-pass backward removes 36 KiB/block scratch at $E=2{,}304$.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
