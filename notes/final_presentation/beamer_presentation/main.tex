\begin{frame}[fragile]{The original top-$k$ path repeatedly rescanned a 2,304-wide row}{Radix selection changes the algorithm from $K$-dependent scans to fixed passes}
  \NvidiaSource{Matched B300 trace-back benchmark; TE PR \href{https://github.com/NVIDIA/TransformerEngine/pull/2821}{\#2821}}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.47\textwidth}
      {\NvidiaSansMedium Naive selection}\par
      \begin{NvidiaCode}
for slot in range(K):
    best = -inf
    for expert in range(E):
        if expert not already_selected:
            best = max(best, score[expert])
      \end{NvidiaCode}
      \centering {\color{NvidiaGray}$\mathcal{O}(E\times K)$-like repeated work}
    \end{column}
    \begin{column}{.47\textwidth}
      {\NvidiaSansMedium Radix selection}\par
      \begin{NvidiaCode}
prefix = 0
for nibble in reversed(range(8)):
    hist[16] = count_matching(scores, prefix)
    bucket = bucket_containing_kth(hist)
    prefix = extend(prefix, bucket)
gather_values_at_or_above(prefix)

      \centering {\color{NvidiaGray}8 fixed FP32 radix passes + gather}
    \end{column}
  \end{columns}

\begin{frame}[fragile]{The optimized router overlaps loads and removes backward scratch}{The win is a coordinated kernel design, not a single launch knob}
  \NvidiaSource{\href{https://github.com/NVIDIA/TransformerEngine/pull/3012}{TE PR \#3012};
    fused-router optimization note}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.51\textwidth}
      \begin{NvidiaCode}
prefetch(tile[0], buffer[0])
for tile in token_tiles:
    wait(buffer[current])
    prefetch(tile[next], buffer[next])

    score(buffer[current])
    radix_topk()
    write_selected()
    swap(current, next)
      \end{NvidiaCode}
    \end{column}
    \begin{column}{.44\textwidth}
      \begin{itemize}
        \item Persistent grid keeps SMs occupied across token rounds.
        \item Double-buffered \texttt{cp.async} overlaps score loads with selection.
        \item Packed histograms cut radix counters from 32 registers to 4; static
          score templates remove hot-path branches.
        \item Two-pass backward removes about 36 KiB/block scratch at $E=2{,}304$.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{B300 roofs show why the router is not a pure bandwidth kernel}{Two deliberately optimistic floors bracket the measured 254.5 $\mu$s result}
  \NvidiaSource{\href{https://www.nvidia.com/en-us/data-center/hgx/}{NVIDIA HGX B300 specs};
    \texttt{TE/.../fused\_router/utils.h}; P3R merged-head benchmark}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.47\textwidth}
      \begin{NvidiaCard}{HBM traffic floor}
        Semantic traffic is approximately 245 MB:
        \[
          245\text{ MB}/8\text{ TB/s}=\alert{30.7\,\mu\text{s}}.
        \]
        The measured kernel reaches about 12\% of raw HBM bandwidth.
      \end{NvidiaCard}
      \vspace{.08in}
      {\scriptsize\color{NvidiaGray}This counts tensor bytes, not cache-line or instruction overhead.}
    \end{column}
    \begin{column}{.47\textwidth}
      \begin{NvidiaCard}{CUDA-core issue floor}
        Eight radix passes touch 151M scores:
        \[
          8\!\times\!8192\!\times\!2304=151\text{M}.
        \]
        One FP32 op/visit at 75 TFLOP/s/GPU gives an impossible
        \alert{$\sim2.0\,\mu$s} floor.
      \end{NvidiaCard}
      \vspace{.08in}
      {\scriptsize\color{NvidiaGray}Real visits include masks, shifts, histograms, shuffles, scans, and dependencies.}
    \end{column}
  \end{columns}
  \vspace{.10in}
\end{frame}

\begin{frame}[fragile]{Compact dense routing removes a 32$\times$ metadata expansion}{Represent selected IDs directly instead of materializing a bool for every expert}
  \NvidiaSource{TE PR \href{https://github.com/NVIDIA/TransformerEngine/pull/3129}{\#3129};
    \texttt{notes/dense\_routing\_rebase\_design.md}}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.48\textwidth}
      \begin{NvidiaCard}{Same choices, smaller state}
        \begin{tabularx}{\linewidth}{@{}X r@{}}
          Bool $[T,E]$ & 2,304 B/token \\
          int16 $[T,K]$ & 72 B/token
        \end{tabularx}
        \[
          \frac{E}{2K}=\frac{2304}{72}=\alert{32\times}.
        \]
      \end{NvidiaCard}
      \vspace{.08in}
      {\small Backward also traverses only the $K$ selected IDs; it no longer asks all $E$
      experts to test a $K$-entry membership list.}
    \end{column}
    \begin{column}{.47\textwidth}
      \begin{NvidiaCode}
route_ids = int16[T, K]
probs = fused_router(logits,
                     dense_route_out=route_ids)

    tokens, dense_route=route_ids, probs=probs
)
      \end{NvidiaCode}
      \vspace{.05in}
      {\small One compact producer-to-consumer contract connects router, scan, dispatch,
      and backward without reconstructing the bool map.}
    \end{column}
  \end{columns}
\end{frame}
