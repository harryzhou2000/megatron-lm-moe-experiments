\section{Fused-Router Optimization}
\NvidiaTransitionSlide{section-router}{Router Optimization}
\NvidiaSectionToc{Fused-Router Optimization}

\begin{frame}{Selection Complexity}{Large $E$ and large $K$ turn repeated selection into the router bottleneck}
  \NvidiaSource{\href{https://github.com/NVIDIA/TransformerEngine/pull/2821}{TE PR \#2821};
    \texttt{TE/.../fused\_router/utils.h}}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.47\textwidth}
      {\NvidiaSansMedium Repeated maximum selection}\par
      \vspace{.04in}{\color{NvidiaGreen}\rule{\linewidth}{.6pt}}\par\vspace{.08in}
      \small A rank-$r$ pass scans $E$ scores and checks up to $r-1$ prior IDs:
      \[
        \sum_{r=1}^{K} E\,r = \alert{O(EK^2)}.
      \]
      \vspace{.05in}
      \small At $E=2{,}304$, $K=36$, selection repeats a full expert-row walk for every
      selected rank.
    \end{column}
    \begin{column}{.47\textwidth}
      {\NvidiaSansMedium Radix selection}\par
      \vspace{.04in}{\color{NvidiaGreen}\rule{\linewidth}{.6pt}}\par\vspace{.08in}
      \small FP32 order bits are narrowed by eight 4-bit passes, followed by a constant
      number of gathers:
      \[
        8E + 2E = \alert{O(E)}.
      \]
      \vspace{.05in}
      \small Work depends on the score representation, not the requested top-$k$. The
      threshold retains the simple path where radix setup is not profitable.
    \end{column}
  \end{columns}
  \vspace{.02in}
  \noindent\NvidiaMetric{2,304}{experts}{one score row per token}\hfill
  \NvidiaMetric{36}{selected IDs}{large sparse fan-out}\hfill
  \NvidiaMetric{8}{radix passes}{FP32, 4 bits/pass}
\end{frame}

\begin{frame}[fragile]{Radix Selection}{Each pass narrows the rank-$K$ score prefix without materializing a sort}
  \NvidiaSource{TE PR \href{https://github.com/NVIDIA/TransformerEngine/pull/2821}{\#2821};
    \texttt{TE/.../fused\_router/utils.h}}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.49\textwidth}
      \begin{NvidiaCode}[basicstyle=\ttfamily\fontsize{7.8}{9.2}\selectfont]
ordered = float_to_ordered_uint(scores)
prefix, mask = 0, 0
rank = topk

for pass in range(7, -1, -1):
    hist[0:16] = count scores matching prefix
    bucket = highest_bucket_containing(rank)
    prefix |= bucket << (4 * pass)
    mask   |= 0xf << (4 * pass)

gather scores > prefix
gather ties in index order until topk
      \end{NvidiaCode}
    \end{column}
    \begin{column}{.45\textwidth}
      {\NvidiaSansMedium Two phases}\par
      \vspace{.04in}{\color{NvidiaGreen}\rule{\linewidth}{.6pt}}\par\vspace{.07in}
      \begin{enumerate}
        \item Count matching scores in 16 buckets; choose the bucket containing rank $K$.
        \item Gather scores above the final value; use ordered ties to complete exactly $K$ IDs.
      \end{enumerate}
      \vspace{.03in}
      \begin{NvidiaEvidence}{Invariant}
        The selected set is exact. Radix changes how the threshold is found, not the top-$k$
        semantics or deterministic tie handling.
      \end{NvidiaEvidence}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Radix Kernel}{The CUDA kernel carries the rank state in registers and uses warp collectives}
  \NvidiaSource{\texttt{TE/transformer\_engine/common/fused\_router/utils.h}}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.58\textwidth}
      \begin{NvidiaCode}[basicstyle=\ttfamily\fontsize{6.5}{7.6}\selectfont]
constexpr int kPasses = 8;
unsigned prefix = 0, mask = 0;
for (int pass = kPasses - 1; pass >= 0; --pass) {
  unsigned packed[4] = {};
  count_matching_scores(packed, prefix, mask);
  int bucket = find_target(packed, k_remaining);
  prefix |= unsigned(bucket) << (4 * pass);
  mask   |= 0xfu << (4 * pass);
}
      \end{NvidiaCode}
    \end{column}
    \begin{column}{.37\textwidth}
      {\NvidiaSansMedium One warp / token}\par
      \vspace{.04in}{\color{NvidiaGreen}\rule{\linewidth}{.6pt}}\par\vspace{.07in}
      \small The implementation maps a token row to a warp. Each lane strides across experts,
      then warp reductions combine 16 bucket counts.
      \vspace{.10in}
      \begin{NvidiaEvidence}{Bounded support}
        Packed 8-bit counters support $E\leq8160$; the 2,304-expert target is well inside that
        range.
      \end{NvidiaEvidence}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Secondary Optimizations}{\href{https://github.com/NVIDIA/TransformerEngine/pull/3012}{PR \#3012} turns the radix algorithm into a throughput-oriented kernel family}
  \NvidiaSource{TE PR \href{https://github.com/NVIDIA/TransformerEngine/pull/3012}{\#3012};
    commits 6d6005a7, 69a7c46f, 00b0dad2, 47c98811, 9cfb651a}
  \vspace{.04in}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.19\textwidth}
      {\NvidiaSansMedium Loop fusion}\par\vspace{.04in}
      \textcolor{NvidiaGreen}{\rule{\linewidth}{.6pt}}\par\vspace{.05in}
      \small Merge clear, load, score, save, and bias work where the score path permits.
    \end{column}
    \begin{column}{.19\textwidth}
      {\NvidiaSansMedium Async load}\par\vspace{.04in}
      \textcolor{NvidiaGreen}{\rule{\linewidth}{.6pt}}\par\vspace{.05in}
      \small Double-buffer raw logits and use a persistent grid sized by occupancy.
    \end{column}
    \begin{column}{.19\textwidth}
      {\NvidiaSansMedium Packed radix}\par\vspace{.04in}
      \textcolor{NvidiaGreen}{\rule{\linewidth}{.6pt}}\par\vspace{.05in}
      \small Store sixteen counters in four 32-bit registers instead of 32 registers.
    \end{column}
    \begin{column}{.19\textwidth}
      {\NvidiaSansMedium Static score}\par\vspace{.04in}
      \textcolor{NvidiaGreen}{\rule{\linewidth}{.6pt}}\par\vspace{.05in}
      \small Instantiate score functions at compile time to remove hot-path branches.
    \end{column}
    \begin{column}{.19\textwidth}
      {\NvidiaSansMedium Merged head}\par\vspace{.04in}
      \textcolor{NvidiaGreen}{\rule{\linewidth}{.6pt}}\par\vspace{.05in}
      \small Keep a specialized small-$k$ head while routing large shapes to the optimized head.
    \end{column}
  \end{columns}
  \vspace{.18in}
  \begin{NvidiaEvidence}{Design principle}
    The algorithmic fix removed the large-$K$ cliff. These refinements remove data movement,
    register, branch, and occupancy costs exposed after that cliff is gone.
  \end{NvidiaEvidence}
\end{frame}

\begin{frame}[fragile]{Loop Fusion}{The score path writes each expert value once instead of staging separate loops}
  \NvidiaSource{TE commit \texttt{6d6005a7};
    \texttt{TE/.../fused\_router/fused\_topk\_with\_score\_function.cu}}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.58\textwidth}
      \begin{NvidiaCode}[basicstyle=\ttfamily\fontsize{6.7}{7.9}\selectfont]
if constexpr (ScoreFunc == kSigmoid) {
  for (int i = lane_id; i < E; i += 32) {
    float value = sigmoid(raw_logits[i]);
    intermediate[pos + i] = value;
    if (bias) value += bias[i];
    scores[i] = value;
  }
} else if constexpr (ScoreFunc == kSqrtsoftplus) {
  for (int i = lane_id; i < E; i += 32) {
    intermediate[pos + i] = raw_logits[i];
    scores[i] = sqrtsoftplus(raw_logits[i]);
  }
}
      \end{NvidiaCode}
    \end{column}
    \begin{column}{.37\textwidth}
      {\NvidiaSansMedium Fused dataflow}\par
      \vspace{.04in}{\color{NvidiaGreen}\rule{\linewidth}{.6pt}}\par\vspace{.07in}
      \small A single strided loop converts logits, evaluates the score function, preserves the
      value required by backward, applies optional bias, and populates selection input.
      \vspace{.10in}
      \small The softmax path remains multi-pass because its normalization is mathematically
      coupled across the entire expert row.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Async Loading}{Raw logits are prefetched with \texttt{cp.async} while the current tile is selected}
  \NvidiaSource{TE commit \texttt{69a7c46f};
    \texttt{TE/.../fused\_router/async\_loader.h}}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.56\textwidth}
      \begin{NvidiaCode}[basicstyle=\ttfamily\fontsize{6.7}{7.9}\selectfont]
// tile n+1 starts before tile n selects
loader.start_load(next_logits, E, lane_id);

loader.wait();
select_topk(loader.current_buf());

// 16-byte global-to-shared copy
cp_async_16B(dst, src);
cp_async_commit();
      \end{NvidiaCode}
    \end{column}
    \begin{column}{.38\textwidth}
      {\NvidiaSansMedium Overlap policy}\par
      \vspace{.04in}{\color{NvidiaGreen}\rule{\linewidth}{.6pt}}\par\vspace{.07in}
      \small The loader keeps the original input dtype in shared memory, deferring conversion to
      compute. It uses 16-byte vector copies on supported GPUs and a scalar fallback otherwise.
      \vspace{.10in}
      \small The host launcher chooses one or two buffers from the shared-memory budget, then
      sizes the persistent grid from active blocks per SM.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Packed Radix}{Eight-bit fields lower histogram bookkeeping from 32 registers to four}
  \NvidiaSource{TE commit \texttt{00b0dad2};
    \texttt{TE/.../fused\_router/utils.h}}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.57\textwidth}
      \begin{NvidiaCode}[basicstyle=\ttfamily\fontsize{6.7}{7.9}\selectfont]
// 16 counters in 4 packed u32 values
unsigned packed[4] = {};
int bucket = (u >> digit_pos) & 0xf;
int pack = bucket >> 2;
int shift = (bucket & 3) << 3;
packed[pack] += 1u << shift;

unsigned count = (packed[b >> 2] >> ((b & 3) << 3)) & 0xffu;
unsigned total = warp_allreduce_sum(count);
      \end{NvidiaCode}
    \end{column}
    \begin{column}{.38\textwidth}
      {\NvidiaSansMedium Why it is safe}\par
      \vspace{.04in}{\color{NvidiaGreen}\rule{\linewidth}{.6pt}}\par\vspace{.07in}
      \small Each lane sees at most $\lceil E/32\rceil$ scores per bucket. For $E=2{,}304$,
      the largest count is 72, which fits in an 8-bit field.
      \vspace{.10in}
      \small The implementation enforces $E\leq8{,}160$ so a packed counter cannot overflow.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Static Scores}{Compile-time score specialization removes the runtime score-function branch}
  \NvidiaSource{TE commit \texttt{47c98811};
    \texttt{TE/.../fused\_router/fused\_topk\_with\_score\_function.cu}}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.55\textwidth}
      \begin{NvidiaCode}[basicstyle=\ttfamily\fontsize{6.5}{7.6}\selectfont]
template <typename T, TopkFuncType TopkFunc, int ScoreFunc>
__global__ void router_forward(...);

if constexpr (ScoreFunc == kSoftmax)
  softmax(scores, E, lane_id);
else if constexpr (ScoreFunc == kSigmoid)
  scores[i] = sigmoid(raw_logits[i]);
else if constexpr (ScoreFunc == kSqrtsoftplus)
  scores[i] = sqrtsoftplus(raw_logits[i]);
}
      \end{NvidiaCode}
    \end{column}
    \begin{column}{.40\textwidth}
      {\NvidiaSansMedium Specialization}\par
      \vspace{.04in}{\color{NvidiaGreen}\rule{\linewidth}{.6pt}}\par\vspace{.07in}
      \small The launcher performs the score-function switch once, then each instantiated
      kernel contains only its required math and synchronization.
      \vspace{.10in}
      \small This benefits both top-$k$ and auxiliary-loss forward/backward paths, where branch
      structure otherwise repeats for every expert.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Merged Head}{A single launcher preserves a lightweight head for small $k$ and the optimized head for large shapes}
  \NvidiaSource{TE commit \texttt{9cfb651a};

  \begin{columns}[T,onlytextwidth]
    \begin{column}{.58\textwidth}
      \begin{NvidiaCode}[basicstyle=\ttfamily\fontsize{6.6}{7.8}\selectfont]
bool use_radix = topk >= radix_threshold && E <= kMaxExperts;

if (!use_radix)
  launch_simple(simple_kernel<...>);
else
  launch(optimized_kernel<..., Radix, ScoreFunc>);
      \end{NvidiaCode}
    \end{column}
    \begin{column}{.37\textwidth}
      {\NvidiaSansMedium Shape-aware dispatch}\par
      \vspace{.04in}{\color{NvidiaGreen}\rule{\linewidth}{.6pt}}\par\vspace{.07in}
      \small The simple head avoids async-loader and persistent-grid overhead on conventional
      small-$k$ shapes. The optimized head is selected only when radix work dominates.
      \vspace{.10in}
      \small This is why gains at $2{,}304/36$ do not require regressing common router shapes.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Dense Backward}{Backward traverses selected IDs directly and zero-fills the required dense gradient}
